+def assert_parses(sql: str, parameters: list | None = None) -> None:
+    try:
+        duckdb.connect().execute(f"EXPLAIN {sql}", parameters or [])
+    except Exception as exc:

+            raise AssertionError(f"{message.splitlines()[0]}\n  in: {sql}") from exc
